01 Reverse a String EASY
Think: Strings are immutable in Python — s[0] = "x" raises TypeError, so you can't swap characters directly. Convert to a list with list(s), swap from both ends inward, then "".join(). Two-pointer approach. (In real code, s[::-1] does it in one step.)
Time: O(n)
Space: O(n)
SOLUTION
from typing import List

def reverse_string(s: str) -> str:
    chars = list(s)  # mutable copy of the characters
    left, right = 0, len(chars) - 1

    while left < right:
        # Swap characters at both ends (tuple unpacking)
        chars[left], chars[right] = chars[right], chars[left]
        left += 1
        right -= 1

    return "".join(chars)

# LeetCode variant modifies in-place (list of chars input), returns None:
def reverse_string_in_place(s: List[str]) -> None:
    left, right = 0, len(s) - 1
    while left < right:
        s[left], s[right] = s[right], s[left]
        left, right = left + 1, right - 1

print(reverse_string("hello"))  # olleh

chars = list("hello")
reverse_string_in_place(chars)
print(chars)                    # ['o', 'l', 'l', 'e', 'h']
print("hello"[::-1])            # olleh  (Pythonic one-liner)
02 Valid Palindrome EASY
Think: Clean the string first (keep only characters where c.isalnum(), then .lower()). Then two pointers from both ends — if ANY mismatch, it's not a palindrome.
Time: O(n)
Space: O(n)
SOLUTION
def is_palindrome(s: str) -> bool:
    # Step 1: Keep only letters and digits, lowercase everything
    clean = "".join(c.lower() for c in s if c.isalnum())

    # Step 2: Two pointers — compare from both ends
    left, right = 0, len(clean) - 1

    while left < right:
        if clean[left] != clean[right]:
            return False  # mismatch!
        left += 1
        right -= 1

    return True  # all matched

print(is_palindrome("A man, a plan, a canal: Panama"))  # True
print(is_palindrome("race a car"))                      # False

# Interview one-liner once cleaned: return clean == clean[::-1]
03 Valid Anagram EASY
Think: Two strings are anagrams if they have the exact same character counts. Count chars in first string (+1), subtract chars from second string (-1). If any count goes below 0, they don't match. In Python, collections.Counter does the counting in one call, and for lowercase-only input a fixed 26-slot list indexed by ord(c) - ord("a") works too.
Time: O(n)
Space: O(1) (fixed alphabet, e.g. 26 letters)
SOLUTION
from collections import Counter, defaultdict

def is_anagram(s: str, t: str) -> bool:
    if len(s) != len(t):
        return False  # quick fail

    count = defaultdict(int)  # missing keys start at 0

    # Count up for s
    for c in s:
        count[c] += 1

    # Count down for t
    for c in t:
        if count[c] == 0:
            return False  # char not in s, or used too many times
        count[c] -= 1

    return True

# Pythonic version — Counter is a dict subclass that counts for you
def is_anagram_counter(s: str, t: str) -> bool:
    return Counter(s) == Counter(t)

# Fixed-size version (lowercase a-z only) — truly O(1) space
def is_anagram_26(s: str, t: str) -> bool:
    if len(s) != len(t):
        return False
    counts = [0] * 26
    for a, b in zip(s, t):
        counts[ord(a) - ord("a")] += 1
        counts[ord(b) - ord("a")] -= 1
    return all(c == 0 for c in counts)

print(is_anagram("anagram", "nagaram"))          # True
print(is_anagram("rat", "car"))                  # False
print(is_anagram_counter("anagram", "nagaram"))  # True
print(is_anagram_26("rat", "car"))               # False
04 Longest Substring Without Repeating Characters MEDIUM
Think: Sliding window. Expand the right side one char at a time. If right side char is already in window (duplicate), shrink from the left until it's gone. Track max window size.
1
Use a set to track characters currently in the window.
2
Expand right (enumerate gives index + char). If char already in the set → shrink from left until it's removed.
3
Update max length after each expansion.
Time: O(n)
Space: O(min(n,m)) m=charset size
SOLUTION
def length_of_longest_substring(s: str) -> int:
    seen = set()
    left = max_len = 0

    for right, ch in enumerate(s):
        # Shrink window until no duplicate
        while ch in seen:
            seen.remove(s[left])  # remove leftmost char
            left += 1

        seen.add(ch)
        max_len = max(max_len, right - left + 1)

    return max_len

# "abcabcbb" → window moves like:
# [a] [ab] [abc] [bca] [cab] [abc] [cb] [b]
#  1    2    3    3     3     3    2   1  → max=3

print(length_of_longest_substring("abcabcbb"))  # 3
print(length_of_longest_substring("pwwkew"))    # 3 ("wke")
05 Group Anagrams MEDIUM
Think: Anagrams have the same characters. If we sort each word, anagrams will produce the same sorted string ("eat"→"aet", "tea"→"aet"). Use the sorted string as a dict key to group them — "".join(sorted(word)), because sorted() on a string returns a list of chars. A defaultdict(list) removes the "create the list if missing" boilerplate.
Time: O(n·m log m) m=word length
Space: O(n)
SOLUTION
from collections import defaultdict
from typing import List

def group_anagrams(strs: List[str]) -> List[List[str]]:
    groups = defaultdict(list)  # sorted_word → [words]

    for word in strs:
        # Sort the word to create a canonical "anagram key"
        key = "".join(sorted(word))  # "eat" → ['a','e','t'] → "aet"
        groups[key].append(word)

    return list(groups.values())

print(group_anagrams(["eat", "tea", "tan", "ate", "nat", "bat"]))
# [['eat', 'tea', 'ate'], ['tan', 'nat'], ['bat']]

# O(n·m) alternative: key = tuple of 26 letter counts (lists aren't hashable, tuples are)
def count_key(word: str) -> tuple:
    counts = [0] * 26
    for c in word:
        counts[ord(c) - ord("a")] += 1
    return tuple(counts)

print(count_key("eat") == count_key("tea"))  # True
06 Longest Palindromic Substring MEDIUM
Think: Expand Around Center. Every palindrome has a center. Try expanding from every possible center (single char for odd-length, gap between two chars for even-length). Track the longest expansion found. The helper returns the palindrome's bounds, so we don't need to mutate outer variables (if you did, a nested function would need nonlocal).
Time: O(n²)
Space: O(1)
SOLUTION — Expand Around Center
from typing import Tuple

def longest_palindrome(s: str) -> str:
    def expand(left: int, right: int) -> Tuple[int, int]:
        # Expand as long as chars match; return bounds of the palindrome found
        while left >= 0 and right < len(s) and s[left] == s[right]:
            left -= 1
            right += 1
        return left + 1, right  # s[left+1:right] is the palindrome

    start, end = 0, 1 if s else 0
    for i in range(len(s)):
        for lo, hi in (expand(i, i),       # odd-length palindromes (center = i)
                       expand(i, i + 1)):  # even-length palindromes (center = gap)
            if hi - lo > end - start:
                start, end = lo, hi

    return s[start:end]  # slicing: end index is exclusive

print(longest_palindrome("babad"))  # bab  ("aba" is also valid)
print(longest_palindrome("cbbd"))   # bb
