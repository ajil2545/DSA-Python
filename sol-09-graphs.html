01Clone GraphMEDIUM
Think: DFS/BFS through the original graph. Use a dict to track already-cloned nodes (old → new). Node objects are hashable by identity, so they work directly as dict keys. For each node, create its clone and recursively clone all neighbors.
Time: O(V+E)
Space: O(V)
SOLUTION
from typing import Dict, List, Optional

class Node:
    def __init__(self, val: int = 0, neighbors: Optional[List["Node"]] = None):
        self.val = val
        self.neighbors = neighbors if neighbors is not None else []

def clone_graph(node: Optional[Node]) -> Optional[Node]:
    if node is None:
        return None

    cloned: Dict[Node, Node] = {}  # original → clone

    def dfs(node: Node) -> Node:
        if node in cloned:
            return cloned[node]    # already cloned

        clone = Node(node.val)
        cloned[node] = clone       # register BEFORE recursing (handles cycles)

        for neighbor in node.neighbors:
            clone.neighbors.append(dfs(neighbor))  # clone each neighbor

        return clone

    return dfs(node)

# 1 — 2
# |   |
# 4 — 3
n1, n2, n3, n4 = Node(1), Node(2), Node(3), Node(4)
n1.neighbors = [n2, n4]
n2.neighbors = [n1, n3]
n3.neighbors = [n2, n4]
n4.neighbors = [n1, n3]

copy = clone_graph(n1)
print(copy is not n1)                        # True — a brand-new object
print(copy.val, [n.val for n in copy.neighbors])  # 1 [2, 4]
print(copy.neighbors[0].neighbors[0] is copy)     # True — cycle preserved in the copy
02Course Schedule (Topological Sort)MEDIUM
Problem: Given prerequisites [a, b] meaning "must take b before a", can you finish all courses?

Think: This is cycle detection in a directed graph. Build adjacency list, then DFS with 3 states: unvisited (0), in progress (1), done (2). If DFS revisits an "in progress" node → cycle detected → return False.
1
Build adjacency list from prerequisites.
2
DFS each course. Mark as "in progress" (1) when entered.
3
If we encounter a node marked "in progress" → cycle! Return False.
4
Mark as "done" (2) when all neighbors are processed. Done nodes are safe.
Time: O(V+E)
Space: O(V+E)
SOLUTION — DFS with 3 states
from typing import List

def can_finish(num_courses: int, prerequisites: List[List[int]]) -> bool:
    # Build adjacency list
    adj = [[] for _ in range(num_courses)]   # NOT [[]] * n — that shares one list!
    for course, prereq in prerequisites:
        adj[prereq].append(course)

    # 0=unvisited, 1=in-progress, 2=done
    state = [0] * num_courses

    def dfs(node: int) -> bool:
        if state[node] == 1:
            return False  # cycle!
        if state[node] == 2:
            return True   # already verified safe

        state[node] = 1   # mark in-progress

        for nxt in adj[node]:
            if not dfs(nxt):
                return False

        state[node] = 2   # mark done
        return True

    return all(dfs(i) for i in range(num_courses))  # all() short-circuits on first False

print(can_finish(2, [[1, 0]]))          # True
print(can_finish(2, [[1, 0], [0, 1]]))  # False (cycle)
ALTERNATIVE — Kahn's BFS (no recursion limit)
from collections import deque
from typing import List

def can_finish_kahn(num_courses: int, prerequisites: List[List[int]]) -> bool:
    adj = [[] for _ in range(num_courses)]
    indegree = [0] * num_courses
    for course, prereq in prerequisites:
        adj[prereq].append(course)
        indegree[course] += 1

    queue = deque(i for i in range(num_courses) if indegree[i] == 0)
    taken = 0
    while queue:
        node = queue.popleft()
        taken += 1
        for nxt in adj[node]:
            indegree[nxt] -= 1
            if indegree[nxt] == 0:
                queue.append(nxt)

    return taken == num_courses   # some course never reached in-degree 0 → cycle

print(can_finish_kahn(2, [[1, 0]]))          # True
print(can_finish_kahn(2, [[1, 0], [0, 1]]))  # False (cycle)
03Pacific Atlantic Water FlowMEDIUM
Think: Instead of simulating water flow from each cell (expensive), reverse it. Start from Pacific edges (top row + left col) and DFS/BFS uphill. Start from Atlantic edges (bottom row + right col) and do the same. Cells reachable from BOTH sets are our answer — in Python that's literally a set intersection: pacific & atlantic.
Time: O(R·C)
Space: O(R·C)
SOLUTION
from typing import List, Set, Tuple

def pacific_atlantic(heights: List[List[int]]) -> List[List[int]]:
    rows, cols = len(heights), len(heights[0])
    pacific: Set[Tuple[int, int]] = set()
    atlantic: Set[Tuple[int, int]] = set()

    def dfs(r, c, visited, prev_height):
        if r < 0 or r >= rows or c < 0 or c >= cols:
            return
        if (r, c) in visited or heights[r][c] < prev_height:
            return
        visited.add((r, c))
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            dfs(r + dr, c + dc, visited, heights[r][c])

    # DFS from all Pacific border cells (left column + top row)
    for r in range(rows):
        dfs(r, 0, pacific, heights[r][0])
    for c in range(cols):
        dfs(0, c, pacific, heights[0][c])

    # DFS from all Atlantic border cells (right column + bottom row)
    for r in range(rows):
        dfs(r, cols - 1, atlantic, heights[r][cols - 1])
    for c in range(cols):
        dfs(rows - 1, c, atlantic, heights[rows - 1][c])

    # Cells reachable from BOTH oceans
    return sorted([r, c] for r, c in pacific & atlantic)

heights = [
    [1, 2, 2, 3, 5],
    [3, 2, 3, 4, 4],
    [2, 4, 5, 3, 1],
    [6, 7, 1, 4, 5],
    [5, 1, 1, 2, 4],
]
print(pacific_atlantic(heights))
# [[0, 4], [1, 3], [1, 4], [2, 2], [3, 0], [3, 1], [4, 0]]
04Word LadderHARD
Problem: Transform begin_word into end_word changing one letter at a time, where every intermediate word must be in word_list. Return the number of words in the shortest sequence (0 if impossible).

Think: Words are nodes; two words are connected if they differ by one letter. "Shortest" in an unweighted graph → BFS. Comparing every pair is O(N²·L), so instead group words into wildcard buckets: "hot" belongs to "*ot", "h*t" and "ho*". All words in one bucket are neighbors.
1
If end_word isn't in the list → return 0 immediately.
2
Build defaultdict(list): pattern → words matching it.
3
BFS from begin_word with (word, steps). For each pattern of the current word, enqueue unvisited words in its bucket.
4
Empty a bucket after scanning it — every word in it is already visited, so it never needs scanning again.
Time: O(N·L²)
Space: O(N·L²)
SOLUTION — BFS with Wildcard Buckets
from collections import defaultdict, deque
from typing import List

def ladder_length(begin_word: str, end_word: str, word_list: List[str]) -> int:
    words = set(word_list)
    if end_word not in words:
        return 0

    L = len(begin_word)
    buckets = defaultdict(list)   # "h*t" → ["hot", "hit", ...]
    for w in words | {begin_word}:
        for i in range(L):
            buckets[w[:i] + "*" + w[i + 1:]].append(w)

    queue = deque([(begin_word, 1)])   # (word, words in sequence so far)
    visited = {begin_word}
    while queue:
        word, steps = queue.popleft()
        if word == end_word:
            return steps
        for i in range(L):
            pattern = word[:i] + "*" + word[i + 1:]
            for nxt in buckets[pattern]:
                if nxt not in visited:
                    visited.add(nxt)
                    queue.append((nxt, steps + 1))
            buckets[pattern] = []   # bucket fully used — never scan it again

    return 0

print(ladder_length("hit", "cog", ["hot", "dot", "dog", "lot", "log", "cog"]))  # 5
print(ladder_length("hit", "cog", ["hot", "dot", "dog", "lot", "log"]))         # 0
# hit → hot → dot → dog → cog  (5 words)
05Alien DictionaryHARD
Problem: Given words sorted in an unknown alphabet, return a valid letter order (any one), or "" if the input is contradictory.

Think: Compare each adjacent pair of words. The first position where they differ tells you one rule: letter in w1 comes before letter in w2 → a directed edge. Then a topological sort of the letters is the alphabet. A cycle means no valid order. Special trap: if w2 is a proper prefix of w1 (e.g. "abc" before "ab"), the input is invalid.
1
Create a node for every letter that appears (even ones with no rules).
2
For each adjacent pair, find the first differing letter → add edge a → b (once). No difference and len(w1) > len(w2) → return "".
3
Kahn's algorithm: start a deque with all letters of in-degree 0, pop, append to order, decrement neighbors.
4
If the order is shorter than the number of letters → cycle → return "".
Time: O(C)
Space: O(U + E)
SOLUTION — Build Graph + Kahn's Topological Sort
from collections import deque
from typing import List

def alien_order(words: List[str]) -> str:
    # 1. Every letter that appears is a node
    graph = {c: set() for w in words for c in w}
    indegree = {c: 0 for c in graph}

    # 2. Adjacent pairs → first differing letter gives one edge
    for w1, w2 in zip(words, words[1:]):
        for a, b in zip(w1, w2):
            if a != b:
                if b not in graph[a]:      # avoid counting a duplicate edge twice
                    graph[a].add(b)
                    indegree[b] += 1
                break                      # only the FIRST difference matters
        else:                            # for-else: loop ended without break
            if len(w1) > len(w2):         # "abc" before "ab" is impossible
                return ""

    # 3. Kahn's BFS (sorted only to make the output deterministic)
    queue = deque(sorted(c for c in graph if indegree[c] == 0))
    order = []
    while queue:
        c = queue.popleft()
        order.append(c)
        for nxt in sorted(graph[c]):
            indegree[nxt] -= 1
            if indegree[nxt] == 0:
                queue.append(nxt)

    # 4. Letters stuck with in-degree > 0 are part of a cycle
    if len(order) < len(graph):
        return ""
    return "".join(order)

print(alien_order(["wrt", "wrf", "er", "ett", "rftt"]))   # wertf
print(alien_order(["z", "x"]))                         # zx
print(repr(alien_order(["z", "x", "z"])))              # ''  (cycle z → x → z)
print(repr(alien_order(["abc", "ab"])))                # ''  (prefix case)
# Rules from "wrt…": t→f, w→e, r→t, e→r  ⇒  w → e → r → t → f
