Linked List Solutions

Pointer manipulation problems — draw diagrams, think in nodes.

01Reverse a Linked ListEASY
Think: Walk through once, flipping each "next" pointer to point backward instead of forward. Use 3 variables: prev (where the current node should point to), curr (node we're processing), nxt (temp save before we overwrite curr.next).
SETUP — ListNode & helpers used by every solution on this page
from typing import List, Optional

class ListNode:
    __slots__ = ("val", "next")

    def __init__(self, val: int = 0, next: Optional["ListNode"] = None):
        self.val = val
        self.next = next

    def __repr__(self) -> str:
        return f"ListNode({self.val})"


def build(values: List[int]) -> Optional[ListNode]:
    """[1, 2, 3] → 1 → 2 → 3 (uses the dummy-node trick)."""
    dummy = tail = ListNode()
    for v in values:
        tail.next = ListNode(v)
        tail = tail.next
    return dummy.next


def to_list(head: Optional[ListNode]) -> List[int]:
    """1 → 2 → 3 → [1, 2, 3] so we can print results."""
    out = []
    while head:
        out.append(head.val)
        head = head.next
    return out
SOLUTION
def reverse_list(head: Optional[ListNode]) -> Optional[ListNode]:
    prev = None   # previous node (new "next" for current)
    curr = head   # node we're currently reversing

    while curr is not None:
        nxt = curr.next    # 1. Save next (we're about to overwrite it)
        curr.next = prev   # 2. Flip pointer backward
        prev = curr        # 3. Move prev forward
        curr = nxt         # 4. Move curr forward

    return prev   # prev is now the new head (last node we processed)

print(to_list(reverse_list(build([1, 2, 3]))))   # [3, 2, 1]

# Step-by-step on 1→2→3→None:
# Step 1: nxt=2,    1.next=None, prev=1, curr=2
# Step 2: nxt=3,    2.next=1,    prev=2, curr=3
# Step 3: nxt=None, 3.next=2,    prev=3, curr=None
# Return prev=3 (new head): 3→2→1→None ✓
02Merge Two Sorted ListsEASY
Think: Use a "dummy" head node so we don't need special cases for the first node. Walk both lists, always pick the smaller node. Attach whatever's left at the end.
SOLUTION
def merge_two_lists(l1: Optional[ListNode], l2: Optional[ListNode]) -> Optional[ListNode]:
    dummy = ListNode()   # placeholder — avoids edge cases
    tail = dummy

    while l1 and l2:
        if l1.val <= l2.val:
            tail.next = l1   # pick from l1
            l1 = l1.next
        else:
            tail.next = l2   # pick from l2
            l2 = l2.next
        tail = tail.next

    # One list is exhausted — attach the rest of the other
    tail.next = l1 or l2     # `or` returns the first truthy operand (or the last one)

    return dummy.next        # skip the dummy node

print(to_list(merge_two_lists(build([1, 2, 4]), build([1, 3, 4]))))
# [1, 1, 2, 3, 4, 4]
03Remove Nth Node From End of ListMEDIUM
Think: To remove the nth from end, we need to find it in one pass. Use two pointers: advance the fast pointer n steps ahead. Then move both until fast hits the end. Now slow is right before the node to remove.
1
Dummy node → makes removing the head easy.
2
Move fast pointer n+1 steps ahead.
3
Move both slow and fast until fast is None.
4
slow.next = slow.next.next (skip the target node).
Time: O(n)
Space: O(1)
SOLUTION
def remove_nth_from_end(head: Optional[ListNode], n: int) -> Optional[ListNode]:
    dummy = ListNode(0, head)
    fast = slow = dummy

    # Move fast n+1 steps ahead
    for _ in range(n + 1):
        fast = fast.next

    # Move both until fast is None
    while fast is not None:
        slow = slow.next
        fast = fast.next

    # slow is the node BEFORE the one to delete
    slow.next = slow.next.next

    return dummy.next

print(to_list(remove_nth_from_end(build([1, 2, 3, 4, 5]), 2)))   # [1, 2, 3, 5]
print(to_list(remove_nth_from_end(build([1]), 1)))               # []  (head removed — dummy handles it)
04Reorder ListMEDIUM
Think: 3 steps — (1) Find the middle using slow/fast pointers. (2) Reverse the second half. (3) Merge the two halves by alternating nodes.
SOLUTION
def reorder_list(head: Optional[ListNode]) -> None:
    """Reorders in place: L0 → Ln → L1 → Ln-1 → ..."""
    if not head:
        return

    # Step 1: Find middle
    slow = fast = head
    while fast.next and fast.next.next:
        slow = slow.next
        fast = fast.next.next

    # Step 2: Reverse second half
    second = slow.next
    slow.next = None   # cut the list
    prev = None
    while second:
        nxt = second.next
        second.next = prev
        prev = second
        second = nxt
    second = prev   # head of the reversed second half

    # Step 3: Merge by alternating nodes
    first = head
    while second:
        tmp1, tmp2 = first.next, second.next
        first.next = second
        second.next = tmp1
        first, second = tmp1, tmp2

head = build([1, 2, 3, 4, 5])
reorder_list(head)
print(to_list(head))   # [1, 5, 2, 4, 3]   i.e. 1→2→3→4→5 becomes 1→5→2→4→3
05Linked List CycleEASY
Problem: Return True if following next pointers ever brings you back to a node you've already visited.

Think: Floyd's tortoise and hare. slow moves 1 step, fast moves 2. With no cycle, fast falls off the end. With a cycle, fast laps slow and they land on the same node object — compare with is (identity), not == (two different nodes can hold equal values).
1
Start slow and fast at head.
2
While fast and fast.next exist: slow = slow.next, fast = fast.next.next.
3
If slow is fast → cycle. If the loop ends → no cycle.
Time: O(n)
Space: O(1)
SOLUTION — Floyd's Tortoise & Hare
def has_cycle(head: Optional[ListNode]) -> bool:
    slow = fast = head
    while fast and fast.next:
        slow = slow.next          # 1 step
        fast = fast.next.next     # 2 steps
        if slow is fast:          # same node OBJECT, not just equal values
            return True
    return False               # fast reached the end → no loop

head = build([3, 2, 0, -4])
print(has_cycle(head))   # False

tail = head
while tail.next:
    tail = tail.next
tail.next = head.next    # -4 → 2 creates a loop (don't call to_list now — it would never end!)
print(has_cycle(head))   # True
print(has_cycle(None))   # False
06Middle of the Linked ListEASY
Think: Same slow/fast trick. fast moves twice as fast, so when fast reaches the end, slow is halfway. For an even length there are two middles — this loop stops on the second one (the LeetCode convention).
Time: O(n)
Space: O(1)
SOLUTION
def middle_node(head: Optional[ListNode]) -> Optional[ListNode]:
    slow = fast = head
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
    return slow   # even length → second middle

print(to_list(middle_node(build([1, 2, 3, 4, 5]))))      # [3, 4, 5]
print(to_list(middle_node(build([1, 2, 3, 4, 5, 6]))))   # [4, 5, 6]
print(middle_node(build([1])))                         # ListNode(1)
07LRU CacheMEDIUM
Problem: Design a cache with a fixed capacity. get(key) returns the value or -1; put(key, value) inserts/updates, and when over capacity evicts the least recently used key. Both must be O(1).

Think: A dict gives O(1) lookup but no notion of "oldest". A doubly linked list gives O(1) move/remove but no lookup. Combine them: dict maps key → node, list keeps nodes in usage order. Python's OrderedDict is exactly this combination built in.
1
On get: if present, move the key to the "most recent" end and return its value.
2
On put: insert/update the key and move it to the "most recent" end.
3
If size > capacity: remove the item at the "least recent" end.
Time: O(1) per get/put
Space: O(capacity)
SOLUTION — OrderedDict (idiomatic Python)
from collections import OrderedDict

class LRUCache:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = OrderedDict()   # key → value, least recent first

    def get(self, key: int) -> int:
        if key not in self.cache:
            return -1
        self.cache.move_to_end(key)       # mark as most recently used
        return self.cache[key]

    def put(self, key: int, value: int) -> None:
        if key in self.cache:
            self.cache.move_to_end(key)
        self.cache[key] = value
        if len(self.cache) > self.capacity:
            self.cache.popitem(last=False)  # evict from the FRONT = least recent

cache = LRUCache(2)
cache.put(1, 1)
cache.put(2, 2)
print(cache.get(1))   # 1
cache.put(3, 3)       # evicts key 2 (key 1 was just used)
print(cache.get(2))   # -1
cache.put(4, 4)       # evicts key 1
print(cache.get(1))   # -1
print(cache.get(3))   # 3
print(cache.get(4))   # 4
SOLUTION — dict + Doubly Linked List with Sentinels (what interviewers want)
class DNode:
    __slots__ = ("key", "val", "prev", "next")

    def __init__(self, key: int = 0, val: int = 0):
        self.key = key       # stored so eviction can delete it from the dict
        self.val = val
        self.prev = None
        self.next = None


class LRUCacheDLL:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.map = {}                # key → DNode
        self.head = DNode()          # sentinel: head.next is MOST recent
        self.tail = DNode()          # sentinel: tail.prev is LEAST recent
        self.head.next = self.tail   # sentinels mean no None checks anywhere
        self.tail.prev = self.head

    def _remove(self, node: DNode) -> None:
        node.prev.next = node.next
        node.next.prev = node.prev

    def _add_front(self, node: DNode) -> None:
        node.prev = self.head
        node.next = self.head.next
        self.head.next.prev = node
        self.head.next = node

    def get(self, key: int) -> int:
        if key not in self.map:
            return -1
        node = self.map[key]
        self._remove(node)          # unlink…
        self._add_front(node)       # …and relink as most recent
        return node.val

    def put(self, key: int, value: int) -> None:
        if key in self.map:
            self._remove(self.map[key])
        node = DNode(key, value)
        self.map[key] = node
        self._add_front(node)
        if len(self.map) > self.capacity:
            lru = self.tail.prev        # least recently used real node
            self._remove(lru)
            del self.map[lru.key]

cache = LRUCacheDLL(2)
cache.put(1, 1)
cache.put(2, 2)
print(cache.get(1))   # 1
cache.put(3, 3)       # evicts key 2
print(cache.get(2))   # -1
cache.put(4, 4)       # evicts key 1
print(cache.get(1))   # -1
print(cache.get(3))   # 3
print(cache.get(4))   # 4
