Stack + Monotonic Stack

A stack is Last-In-First-Out. In Python, a plain list with append() and pop() is a perfect O(1) stack; peek with stack[-1] and test emptiness with if stack:. The monotonic stack is a stack that stays sorted — it's the key to "next greater/smaller element" problems.

Problem 1: Valid Parentheses

Given a string of brackets (){}[], determine if the input is valid.

valid_parentheses.py 
# O(n) time, O(n) space
def is_valid(s: str) -> bool:
    stack = []
    pairs = {')': '(', '}': '{', ']': '['}

    for c in s:
        if c in pairs:
            # Closing bracket — top of stack must match.
            # Guard first: pop() on an empty list raises IndexError.
            if not stack or stack.pop() != pairs[c]:
                return False
        else:
            stack.append(c)  # Opening bracket — push onto stack
    return not stack          # valid only if nothing is left open

print(is_valid("()[]{}"))   # True
print(is_valid("(]"))       # False
print(is_valid("]"))        # False — no crash on empty stack

# Pattern: Stack for matching/nesting
# Trigger: brackets, nested structures, undo operations

Problem 2: Daily Temperatures (Monotonic Stack)

Given daily temperatures, for each day find how many days until a warmer day. Return 0 if no warmer day.

daily_temperatures.py 
# LAYER 1: Brute force — O(n²), check every future day
# LAYER 2: Monotonic Decreasing Stack — O(n) time, O(n) space

def daily_temperatures(temps):
    result = [0] * len(temps)   # flat list of ints — [0] * n is safe here
    stack = []                  # stores INDICES (not values!)

    for i, t in enumerate(temps):
        # Pop all days colder than today — today is their answer
        while stack and t > temps[stack[-1]]:
            prev_day = stack.pop()
            result[prev_day] = i - prev_day
        stack.append(i)
    return result

print(daily_temperatures([73, 74, 75, 71, 69, 72, 76, 73]))
# [1, 1, 4, 2, 1, 1, 0, 0]

# KEY INSIGHT: the stack maintains indices in decreasing temp order.
# When a warmer day arrives, it "resolves" all colder days on the stack.
# Each element is pushed and popped at most once → O(n) total.

# LAYER 3: Pattern — Monotonic Decreasing Stack
# Trigger: "next greater element", "days until warmer/higher"
# Stack stores: INDICES (not values) in decreasing order
# Pop condition: current > stack top

Problem 3: Largest Rectangle in Histogram

Given histogram bar heights, find the largest rectangle area. This is the "boss fight" of monotonic stack problems.

largest_rectangle_histogram.py 
# O(n) time, O(n) space — Monotonic Increasing Stack
def largest_rectangle_area(heights):
    stack = []        # stores indices, heights in increasing order
    max_area = 0
    n = len(heights)

    for i in range(n + 1):
        h = 0 if i == n else heights[i]   # sentinel

        while stack and h < heights[stack[-1]]:
            height = heights[stack.pop()]
            # Width: from current index to the new stack top
            width = i - stack[-1] - 1 if stack else i
            max_area = max(max_area, height * width)
        stack.append(i)
    return max_area

print(largest_rectangle_area([2, 1, 5, 6, 2, 3]))   # 10

# WHY IT WORKS: for each bar, we find the furthest it can extend
# left and right without hitting a shorter bar. The stack gives us
# both boundaries in amortized O(1) by maintaining increasing order.
# Sentinel (height 0 at end) flushes all remaining bars.

Linked Lists in Python

Python has no built-in linked list you can manipulate node by node (collections.deque is linked internally but hides its nodes). Building one teaches pointer manipulation — the exact skill you need for trees and graphs. A "pointer" in Python is just a name referencing a node object, exactly as you saw on Day 1. Focus on the pointer redirection pattern: always draw it out before coding.

linked_list_node.py 
from typing import Optional, List

# The building block — memorize this (LeetCode's exact definition)
class ListNode:
    def __init__(self, val=0, next=None):   # `next` shadows a built-in; fine as an attribute
        self.val = val
        self.next = next

# Test helpers — build a list from Python values and back
def build(values: List[int]) -> Optional[ListNode]:
    dummy = ListNode()
    tail = dummy
    for v in values:
        tail.next = ListNode(v)
        tail = tail.next
    return dummy.next

def to_list(head: Optional[ListNode]) -> List[int]:
    out = []
    while head:
        out.append(head.val)
        head = head.next
    return out

print(to_list(build([1, 2, 3])))   # [1, 2, 3]

Problem 1: Reverse Linked List

The single most important linked list operation. If you can reverse a list, you can handle 90% of LL problems.

reverse_linked_list.py 
# Uses ListNode, build, to_list from linked_list_node.py

# Iterative — O(n) time, O(1) space
def reverse_list(head):
    prev = None
    curr = head

    while curr:
        nxt = curr.next      # save next
        curr.next = prev     # reverse pointer
        prev = curr          # advance prev
        curr = nxt           # advance curr
    return prev  # prev is the new head

# Pythonic one-liner body (right side is evaluated BEFORE any assignment):
#     curr.next, prev, curr = prev, curr, curr.next

# Recursive — O(n) time, O(n) space (call stack; Python's default limit is ~1000 frames)
def reverse_list_recursive(head):
    if not head or not head.next:
        return head
    new_head = reverse_list_recursive(head.next)
    head.next.next = head  # reverse the link
    head.next = None       # break old link
    return new_head

print(to_list(reverse_list(build([1, 2, 3, 4, 5]))))            # [5, 4, 3, 2, 1]
print(to_list(reverse_list_recursive(build([1, 2, 3]))))        # [3, 2, 1]

# MANTRA: save → reverse → advance → advance
# Draw the arrows on paper. Every time.

Problem 2: Merge Two Sorted Lists

Merge two sorted linked lists into one sorted list.

merge_two_sorted_lists.py 
# O(n + m) time, O(1) space
def merge_two_lists(l1, l2):
    dummy = ListNode(0)  # dummy head avoids edge cases
    tail = dummy

    while l1 and l2:
        if l1.val <= l2.val:
            tail.next = l1
            l1 = l1.next
        else:
            tail.next = l2
            l2 = l2.next
        tail = tail.next
    tail.next = l1 or l2  # attach remaining (`or` returns the first truthy operand)
    return dummy.next

print(to_list(merge_two_lists(build([1, 2, 4]), build([1, 3, 4]))))
# [1, 1, 2, 3, 4, 4]

# KEY TRICK: the dummy node. It eliminates "is head None?" checks.
# You'll use this pattern in most LL construction problems.

Problem 3: Linked List Cycle (Fast & Slow Pointers)

Determine if a linked list has a cycle.

linked_list_cycle.py 
# Floyd's Tortoise and Hare — O(n) time, O(1) space
def has_cycle(head):
    slow = fast = head

    while fast and fast.next:
        slow = slow.next        # 1 step
        fast = fast.next.next   # 2 steps
        if slow is fast:         # same NODE object → cycle (identity, not ==)
            return True
    return False  # fast hit None → no cycle

# BONUS: Find WHERE the cycle starts
def detect_cycle(head):
    slow = fast = head

    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
        if slow is fast:
            # Reset one pointer to head, move both at speed 1
            entry = head
            while entry is not slow:
                entry = entry.next
                slow = slow.next
            return entry  # cycle start node
    return None

head = build([3, 2, 0, -4])
print(has_cycle(head))              # False
head.next.next.next.next = head.next  # tail (-4) → node 2
print(has_cycle(head))              # True
print(detect_cycle(head).val)       # 2

# Pattern: Fast & Slow Pointers (Floyd's)
# Trigger: "cycle detection", "find middle node", "kth from end"
# Why it works: if there's a cycle, fast gains 1 step per iteration,
# so they must eventually meet. Math guarantees it.
# (The O(n)-space alternative: a set of seen nodes — nodes are hashable by identity.)

Prefix Sum + Merge Intervals

PATTERN TRIGGER "Range sum query" or "subarray sum equals K" → Prefix Sum. "Overlapping intervals/scheduling" → Sort + Merge.

Problem 1: Range Sum Query

Given an array, answer multiple queries: "what's the sum from index i to j?"

prefix_sum.py 
from itertools import accumulate

# Prefix Sum Array — O(n) build, O(1) per query
class NumArray:
    def __init__(self, nums):
        self.prefix = [0]  # prefix[0] = 0 as sentinel
        for n in nums:
            self.prefix.append(self.prefix[-1] + n)
        # Pythonic equivalent: self.prefix = [0] + list(accumulate(nums))

    def sum_range(self, i, j):
        return self.prefix[j + 1] - self.prefix[i]

na = NumArray([-2, 0, 3, -5, 2, -1])
print(na.sum_range(0, 2))   # 1
print(na.sum_range(2, 5))   # -1
print([0] + list(accumulate([1, 2, 3])))   # [0, 1, 3, 6]

# prefix = [0, a, a+b, a+b+c, ...]
# sum(i..j) = prefix[j+1] - prefix[i]
# The extra 0 at the start avoids edge cases.
# (sum(nums[i:j+1]) per query would be O(n) each — that's what we're avoiding.)

Problem 2: Subarray Sum Equals K

Count the number of contiguous subarrays that sum to k. This combines prefix sum with hash map — a powerful fusion.

subarray_sum_k.py 
from collections import defaultdict

# O(n) time, O(n) space — Prefix Sum + Hash Map combo
def subarray_sum(nums, k):
    prefix_counts = defaultdict(int)
    prefix_counts[0] = 1   # empty prefix has sum 0
    running = count = 0

    for num in nums:
        running += num
        # If (running - k) was seen before, there's a subarray summing to k.
        # .get() reads without inserting a key (plain [] on a defaultdict would insert 0)
        count += prefix_counts.get(running - k, 0)
        prefix_counts[running] += 1
    return count

print(subarray_sum([1, 1, 1], 2))   # 2
print(subarray_sum([1, 2, 3], 3))   # 2

# WHY IT WORKS:
# If prefix[j] - prefix[i] = k, then subarray [i+1..j] sums to k.
# Rearranged: prefix[i] = prefix[j] - k
# So for each j, count how many previous prefix sums equal (running - k).
# This is the Two Sum pattern applied to prefix sums!

Problem 3: Merge Intervals

Given a collection of intervals, merge all overlapping intervals.

merge_intervals.py 
# O(n log n) time (sorting), O(n) space
def merge(intervals):
    if not intervals:
        return []
    intervals.sort(key=lambda iv: iv[0])   # sort by start
    merged = [list(intervals[0])]          # copy, so we never mutate the caller's lists

    for start, end in intervals[1:]:
        last = merged[-1]
        if start <= last[1]:
            # Overlapping — extend the end
            last[1] = max(last[1], end)
        else:
            # No overlap — add new interval
            merged.append([start, end])
    return merged

print(merge([[1, 3], [2, 6], [8, 10], [15, 18]]))   # [[1, 6], [8, 10], [15, 18]]
print(merge([[1, 4], [4, 5]]))                      # [[1, 5]]

# Pattern: Sort + Linear Scan
# Trigger: "overlapping ranges", "meeting rooms", "scheduling"
# Always sort by start time first. Then single pass to merge.
# Day 1 callback: merged = [intervals[0]] would ALIAS the input's first
# interval, and last[1] = ... would silently edit the caller's data.

Pattern Fusions to Document

Prefix Sum + Hash Map = Subarray Sum Equals K. The hash map stores prefix sum frequencies, turning a nested loop into a single pass. This is literally Two Sum applied to running totals.

Monotonic Stack + Array = Next Greater Element. The stack maintains a decreasing sequence; each pop resolves a "waiting" element.

Sort + Two Pointers = 3Sum, Merge Intervals. Sorting creates structure that pointers or linear scans can exploit.

These fusions are where real problem-solving power comes from. A single pattern solves simple problems. Pattern combinations solve hard problems.

LeetCode 238 — Product of Array Except Self

No division allowed. This is a prefix/suffix product problem — a twist on prefix sums.

product_except_self.py 
# O(n) time, O(1) extra space (output list doesn't count)
def product_except_self(nums):
    n = len(nums)
    result = [1] * n

    # Pass 1: left products (prefix product)
    for i in range(1, n):
        result[i] = result[i - 1] * nums[i - 1]

    # Pass 2: multiply by right products (suffix product)
    right_product = 1
    for i in reversed(range(n)):   # same as range(n - 1, -1, -1)
        result[i] *= right_product
        right_product *= nums[i]
    return result

print(product_except_self([1, 2, 3, 4]))        # [24, 12, 8, 6]
print(product_except_self([-1, 1, 0, -3, 3]))   # [0, 0, 9, 0, 0]

# KEY INSIGHT: product[i] = (product of everything left) × (product of everything right)
# Two passes: left→right builds prefix products, right→left multiplies suffix products.
# Pattern: Prefix/Suffix decomposition. Extends to any associative operation.

Queue + Priority Queue (heapq)

Python ships a binary heap in the standard library: heapq, a set of functions that treat a plain list as a min-heap. It is what you'll use in interviews. To understand what it does, read the hand-rolled version below first, then try rewriting it from memory.

MinHeap Implementation (for later — understand the API now)

A min-heap gives O(log n) insert/remove and O(1) peek at the minimum (a max-heap does the same for the maximum). For DSA problems, you mostly need: push(val), pop() (remove min), peek(). With heapq those are heappush(h, x), heappop(h), and h[0]. There is no max-heap: push negated values instead.

min_heap.py 
import heapq

# MinHeap — complete implementation for DSA
class MinHeap:
    def __init__(self):
        self.data = []

    def push(self, val):
        self.data.append(val)
        self._bubble_up(len(self.data) - 1)

    def pop(self):
        smallest = self.data[0]
        last = self.data.pop()
        if self.data:
            self.data[0] = last
            self._sink_down(0)
        return smallest

    def peek(self):
        return self.data[0]

    def __len__(self):          # enables len(heap) and `if heap:`
        return len(self.data)

    def _bubble_up(self, i):
        data = self.data
        while i > 0:
            parent = (i - 1) // 2
            if data[i] >= data[parent]:
                break
            data[i], data[parent] = data[parent], data[i]   # tuple swap
            i = parent

    def _sink_down(self, i):
        data, n = self.data, len(self.data)
        while True:
            smallest = i
            l, r = 2 * i + 1, 2 * i + 2
            if l < n and data[l] < data[smallest]:
                smallest = l
            if r < n and data[r] < data[smallest]:
                smallest = r
            if smallest == i:
                break
            data[i], data[smallest] = data[smallest], data[i]
            i = smallest

# Usage:
heap = MinHeap()
heap.push(5); heap.push(2); heap.push(8)
print(heap.pop())    # 2 (smallest)
print(heap.peek())   # 5 (new smallest)

# The same thing with the standard library:
h = []
for x in (5, 2, 8):
    heapq.heappush(h, x)
print(heapq.heappop(h), h[0])        # 2 5

# Max-heap trick: negate on the way in and out
mx = [-x for x in (5, 2, 8)]
heapq.heapify(mx)                     # O(n) build
print(-heapq.heappop(mx))             # 8 (largest)

Problem: K Closest Points to Origin

Given points array, find k closest to origin (0,0).

k_closest_points.py 
import heapq

# Approach 1: Sort — O(n log n)
def k_closest_sort(points, k):
    return sorted(points, key=lambda p: p[0] ** 2 + p[1] ** 2)[:k]

# Approach 2: MaxHeap of size k — O(n log k)
# Keep a max-heap of size k. For each point, if its distance
# is smaller than the heap's max, replace. Final heap = k closest.
# (heapq is a MinHeap, so store NEGATED distances to simulate a MaxHeap)
def k_closest(points, k):
    heap = []   # entries: (-dist, x, y)
    for x, y in points:
        d = -(x * x + y * y)
        if len(heap) < k:
            heapq.heappush(heap, (d, x, y))
        elif d > heap[0][0]:               # closer than the farthest kept
            heapq.heapreplace(heap, (d, x, y))  # pop + push in one O(log k) step
    return [[x, y] for _, x, y in heap]

print(k_closest_sort([[1, 3], [-2, 2]], 1))                  # [[-2, 2]]
print(sorted(k_closest([[3, 3], [5, -1], [-2, 4]], 2)))   # [[-2, 4], [3, 3]]

# One-liner from the standard library (also O(n log k)):
print(heapq.nsmallest(1, [[1, 3], [-2, 2]], key=lambda p: p[0] ** 2 + p[1] ** 2))   # [[-2, 2]]
