Day 1 · 90 min

Python Memory Model + Complexity Mindset

Before you write another solution, you need to know the true cost of every operation in Python and exactly what a variable holds. Most bugs and performance issues in Python DSA come from not understanding what CPython does under the hood.

Everything Is an Object — Names Are References

In Python a variable is a name bound to an object. Assignment never copies; it just adds another name for the same object. What matters is whether the object is immutable (int, float, bool, str, tuple, frozenset, None) or mutable (list, dict, set, deque, your own class instances). Mutating a shared mutable object is visible through every name.

== asks "same value?", is asks "same object?". Use is only for None (and sentinels); use == for everything else.

names_and_references.py 
# IMMUTABLE — rebinding a name never affects other names
a = 5
b = a          # b names the SAME int object as a
b = 10         # rebinds b to a new object; a is untouched
print(a)       # 5

# MUTABLE — every name sees the mutation (dangerous in DSA!)
arr1 = [1, 2, 3]
arr2 = arr1    # arr2 points to the SAME list
arr2.append(4)
print(arr1)    # [1, 2, 3, 4] — mutated!

# is vs ==  →  identity vs equality
print(arr1 == [1, 2, 3, 4])   # True  — same contents
print(arr1 is [1, 2, 3, 4])   # False — different objects
print(arr1 is arr2)           # True  — one object, two names
x = None
print(x is None)              # True  — the idiomatic None check

# Functions receive references too ("pass by object reference")
def add_item(lst):
    lst.append(99)    # mutates the caller's list

def rebind(lst):
    lst = [0]         # only rebinds the LOCAL name — caller unaffected

nums = [1]
add_item(nums)
rebind(nums)
print(nums)          # [1, 99]

Shallow vs Deep Copy — and the Three Classic Traps

lst[:], list(lst) and lst.copy() make a shallow copy: a new outer list whose slots still point at the same inner objects. copy.deepcopy copies recursively. Three bugs every Python DSA learner hits: the [[0] * n] * m grid, the mutable default argument, and appending a path in backtracking without copying it.

copies_and_traps.py 
import copy

nested = [[1, 2], [3, 4]]
shallow = nested[:]               # shallow copy (same as list(nested) / nested.copy())
deep = copy.deepcopy(nested)     # deep copy — inner lists copied too

nested[0].append(99)
print(shallow)   # [[1, 2, 99], [3, 4]] — inner lists are shared!
print(deep)      # [[1, 2], [3, 4]]     — fully independent

# TRAP 1: [[0] * n] * m  →  m references to ONE row
grid = [[0] * 3] * 2
grid[0][0] = 1
print(grid)      # [[1, 0, 0], [1, 0, 0]] — both rows changed!

grid = [[0] * 3 for _ in range(2)]   # FIX: a fresh row per iteration
grid[0][0] = 1
print(grid)      # [[1, 0, 0], [0, 0, 0]]

# TRAP 2: default arguments are evaluated ONCE, at def time
def bad_append(x, acc=[]):
    acc.append(x)
    return acc

print(bad_append(1))   # [1]
print(bad_append(2))   # [1, 2] — the same list is reused!

def good_append(x, acc=None):
    if acc is None:
        acc = []
    acc.append(x)
    return acc

print(good_append(2))  # [2]

# TRAP 3: backtracking — store a SNAPSHOT of the path
res, path = [], [1, 2]
res.append(path)       # WRONG: keeps a live reference
res.append(path[:])    # RIGHT: copy of the current state
path.pop()
print(res)             # [[1], [1, 2]] — the first entry changed under you

PYTHON TRAP: Using list.pop(0) as a queue If you use a list as a queue with pop(0), every dequeue is O(n). For BFS on large graphs, this silently turns your O(V+E) algorithm into O(V²). Use collections.deque and popleft() — always.
efficient_queue.py 
from collections import deque

# BAD: O(n) per dequeue
queue = [1, 2, 3]
queue.append(4)     # enqueue
queue.pop(0)        # dequeue — O(n)! shifts every element

# GOOD: O(1) at both ends — collections.deque
q = deque([1, 2, 3])
q.append(4)         # enqueue on the right
print(q.popleft())  # 1 — O(1) dequeue
q.appendleft(0)     # O(1) push to the front
print(q)            # deque([0, 2, 3, 4])

# A thin Queue wrapper (you rarely need one — deque IS the queue)
class Queue:
    def __init__(self):
        self._items = deque()

    def enqueue(self, val):
        self._items.append(val)

    def dequeue(self):
        return self._items.popleft()

    def __len__(self):
        return len(self._items)

    def is_empty(self):
        return not self._items   # empty deque is falsy

qq = Queue()
qq.enqueue('a')
qq.enqueue('b')
print(qq.dequeue(), len(qq), qq.is_empty())   # a 1 False

dict vs defaultdict vs Counter — When to Use Which

A dict key can be any hashable (immutable) object: int, str, tuple, frozenset — never a list. Keys are not coerced (1 and '1' are different keys), and insertion order is preserved. Reach for defaultdict when every missing key needs a starting value, and Counter whenever you are counting.

dict_defaultdict_counter.py 
from collections import defaultdict, Counter

# dict: any HASHABLE key, no type coercion
d = {}
d[1] = 'number one'
d['1'] = 'string one'
print(len(d))              # 2 — no collision
d[(0, 1)] = 'cell'          # tuples make perfect grid-coordinate keys
# d[[0, 1]] = 'x'          # TypeError: unhashable type: 'list'

# GOTCHA: 1, 1.0 and True are equal AND hash the same → one key
print({1: 'a', 1.0: 'b', True: 'c'})   # {1: 'c'}

# FREQUENCY COUNTER — the #1 DSA pattern, three ways
def freq_count(arr):
    freq = {}
    for item in arr:
        freq[item] = freq.get(item, 0) + 1
    return freq

print(freq_count("banana"))     # {'b': 1, 'a': 3, 'n': 2}

freq = defaultdict(int)           # missing keys start at int() == 0
for ch in "banana":
    freq[ch] += 1
print(dict(freq))                # {'b': 1, 'a': 3, 'n': 2}

c = Counter("banana")               # the one-liner
print(c.most_common(1))          # [('a', 3)]
print(c['z'])                   # 0 — missing keys read as 0, no KeyError

Truthiness, Big Integers & Immutable Strings

Truthiness: 0, 0.0, '', None, and empty containers are falsy — so while stack: and if not queue: are idiomatic. But a valid value of 0 is also falsy: use is not None when 0 is legal. Integers are arbitrary precision — no overflow, ever (but // floors toward negative infinity). Strings are immutable: convert to a list to edit, and build results with "".join().

Hash Map Pattern

The hash map is the most versatile DSA tool in Python. The pattern: trade space for time — use a dict/set to turn O(n) lookups into O(1).

PATTERN TRIGGER "Find a pair / complement / duplicate / frequency / group by" → reach for a Hash Map.

Problem 1: Two Sum

Given an array of integers and a target, return indices of two numbers that add up to the target. Each input has exactly one solution.

two_sum.py — 3-Layer Method 
from typing import List

# LAYER 1: Brute Force — O(n²) time, O(1) space
def two_sum_brute(nums: List[int], target: int) -> List[int]:
    n = len(nums)
    for i in range(n):
        for j in range(i + 1, n):
            if nums[i] + nums[j] == target:
                return [i, j]
    return []

# BOTTLENECK: inner loop searches for complement linearly
# KEY INSIGHT: store seen numbers in a dict for O(1) lookup

# LAYER 2: Hash Map — O(n) time, O(n) space
def two_sum(nums: List[int], target: int) -> List[int]:
    seen = {}  # value → index
    for i, num in enumerate(nums):
        complement = target - num
        if complement in seen:
            return [seen[complement], i]
        seen[num] = i
    return []

print(two_sum_brute([2, 7, 11, 15], 9))  # [0, 1]
print(two_sum([3, 2, 4], 6))              # [1, 2]

# LAYER 3: Pattern Journal
# Pattern: Hash Map Complement Lookup
# Trigger: "find pair that sums to target"
# Trade-off: O(n) space buys O(n) time (from O(n²))
# Related: 3Sum (sort + two pointers), 4Sum

Problem 2: Group Anagrams

Given an array of strings, group the anagrams together. Two strings are anagrams if they have the same characters with the same frequency.

group_anagrams.py 
from collections import defaultdict

# LAYER 1: Brute Force — compare every pair O(n² · k log k)
# Skip — wasteful. Jump to the real insight.

# LAYER 2: Sort as Key — O(n · k log k) time, O(n·k) space
def group_anagrams(strs):
    groups = defaultdict(list)
    for s in strs:
        key = "".join(sorted(s))   # "eat" → "aet"
        groups[key].append(s)
    return list(groups.values())

# LAYER 2b: Frequency Key — O(n · k) time (no sorting!)
def group_anagrams_freq(strs):
    groups = defaultdict(list)
    for s in strs:
        count = [0] * 26
        for c in s:
            count[ord(c) - ord('a')] += 1
        groups[tuple(count)].append(s)   # lists aren't hashable → tuple
    return list(groups.values())

words = ["eat", "tea", "tan", "ate", "nat", "bat"]
print(group_anagrams(words))       # [['eat', 'tea', 'ate'], ['tan', 'nat'], ['bat']]
print(group_anagrams_freq(words))  # [['eat', 'tea', 'ate'], ['tan', 'nat'], ['bat']]

# LAYER 3: Pattern — Hash Map Grouping
# Trigger: "group items by shared property"
# Key trick: canonical form as dict key (sorted string or count tuple)
# Trade-off: O(n·k) space; freq key avoids k log k sorting cost

Problem 3: Top K Frequent Elements

Given an integer array and integer k, return the k most frequent elements. Guaranteed unique answer.

top_k_frequent.py 
from collections import Counter

# LAYER 1: Sort by frequency — O(n log n)
def top_k_frequent_sort(nums, k):
    freq = Counter(nums)
    ranked = sorted(freq.items(), key=lambda e: e[1], reverse=True)
    return [num for num, _ in ranked[:k]]
    # Interview-ready shortcut: [num for num, _ in freq.most_common(k)]

# BOTTLENECK: sorting is O(n log n). Can we do O(n)?

# LAYER 2: Bucket Sort — O(n) time, O(n) space
def top_k_frequent(nums, k):
    freq = Counter(nums)

    # Bucket: index = frequency, value = list of numbers
    # NOT [[]] * (n + 1) — that would be n+1 references to ONE list
    buckets = [[] for _ in range(len(nums) + 1)]
    for num, count in freq.items():
        buckets[count].append(num)

    # Walk buckets from highest frequency
    result = []
    for i in range(len(buckets) - 1, 0, -1):
        for num in buckets[i]:
            result.append(num)
            if len(result) == k:
                return result
    return result

print(top_k_frequent_sort([1, 1, 1, 2, 2, 3], 2))  # [1, 2]
print(top_k_frequent([1, 1, 1, 2, 2, 3], 2))       # [1, 2]

# LAYER 3: Pattern — Frequency Map + Bucket Sort
# Trigger: "top k" / "k most frequent" / "k most common"
# Key insight: bucket index = frequency, walk backwards
# Trade-off: O(n) space for bucket array; O(n) time total

Two Pointers Pattern

Two pointers eliminates an inner loop by using the structure of the data (usually sorted). One pointer starts left, one right — they walk toward each other based on conditions. This turns O(n²) into O(n).

PATTERN TRIGGER "Sorted array + find pair/triplet" or "palindrome check" or "partition" → Two Pointers.

Problem 1: Valid Palindrome

Given a string, determine if it's a palindrome considering only alphanumeric characters, ignoring case.

valid_palindrome.py 
# LAYER 1: Clean + Reverse — O(n) time, O(n) space
def is_palindrome_brute(s: str) -> bool:
    clean = [c.lower() for c in s if c.isalnum()]
    return clean == clean[::-1]

# LAYER 2: Two Pointers — O(n) time, O(1) space
def is_palindrome(s: str) -> bool:
    l, r = 0, len(s) - 1
    while l < r:
        while l < r and not s[l].isalnum():
            l += 1
        while l < r and not s[r].isalnum():
            r -= 1
        if s[l].lower() != s[r].lower():
            return False
        l, r = l + 1, r - 1
    return True

print(is_palindrome_brute("A man, a plan, a canal: Panama"))  # True
print(is_palindrome("race a car"))                        # False

# LAYER 3: Pattern — Inward Two Pointers
# Trigger: "is palindrome" / "compare from both ends"
# Why O(1) space: no reversed copy needed (s[::-1] would be O(n))
# Template: l = 0, r = end, move inward, compare

Problem 2: Three Sum (3Sum)

Given array nums, find all unique triplets [a, b, c] where a + b + c = 0.

three_sum.py 
# LAYER 2: Sort + Two Pointers — O(n²) time, O(1) extra space
# (Brute force is O(n³) — 3 nested loops. Skip.)

def three_sum(nums):
    nums.sort()           # in-place sort — O(n log n)
    result = []
    n = len(nums)

    for i in range(n - 2):
        # Skip duplicate first element
        if i > 0 and nums[i] == nums[i - 1]:
            continue

        l, r = i + 1, n - 1
        while l < r:
            total = nums[i] + nums[l] + nums[r]   # don't shadow built-in sum()
            if total == 0:
                result.append([nums[i], nums[l], nums[r]])
                # Skip duplicates
                while l < r and nums[l] == nums[l + 1]:
                    l += 1
                while l < r and nums[r] == nums[r - 1]:
                    r -= 1
                l, r = l + 1, r - 1
            elif total < 0:
                l += 1   # need bigger sum → move left pointer right
            else:
                r -= 1   # need smaller sum → move right pointer left
    return result

print(three_sum([-1, 0, 1, 2, -1, -4]))   # [[-1, -1, 2], [-1, 0, 1]]

# LAYER 3: Pattern — Fix one + Two Pointer sweep
# Trigger: "find triplet summing to X" in unsorted array
# Key trick: sort first → enables two-pointer on remainder
# Duplicate skip: the most common bug source. Handle explicitly.

Problem 3: Container With Most Water

Given heights array, find two lines that form a container holding the most water.

container_water.py 
# LAYER 2: Two Pointers — O(n) time, O(1) space
# Brute force O(n²) checks all pairs. We can do better.
# KEY INSIGHT: start wide, move the shorter side inward.
# Why? Moving the taller side can only decrease width
# while the height is still capped by the shorter side.

def max_area(height):
    l, r = 0, len(height) - 1
    best = 0                # don't shadow built-in max()

    while l < r:
        area = min(height[l], height[r]) * (r - l)
        best = max(best, area)

        if height[l] < height[r]:
            l += 1   # move the shorter side
        else:
            r -= 1
    return best

print(max_area([1, 8, 6, 2, 5, 4, 8, 3, 7]))   # 49

# LAYER 3: Pattern — Greedy Two Pointers
# Trigger: "maximize area/product between two positions"
# Why it works: moving shorter side is the only move
# that can increase the area. Greedy + proof by elimination.

Sliding Window Pattern

Sliding window maintains a "window" over a contiguous subarray or substring and slides it forward. Two types: fixed-size (window = k) and variable-size (window expands/contracts based on a condition).

PATTERN TRIGGER "Subarray of size k" or "longest/shortest substring with condition" → Sliding Window.

Problem 1: Max Sum Subarray of Size K (Fixed Window)

Find the maximum sum of any contiguous subarray of size k.

max_sum_k.py 
# LAYER 2: Fixed Sliding Window — O(n) time, O(1) space
def max_sum_subarray(arr, k):
    window_sum = sum(arr[:k])     # build first window — O(k), once
    best = window_sum

    # Slide: add right, remove left
    for i in range(k, len(arr)):
        window_sum += arr[i] - arr[i - k]
        best = max(best, window_sum)
    return best

print(max_sum_subarray([2, 1, 5, 1, 3, 2], 3))   # 9

# Template: fixed window is always this shape.
# Build window [0..k-1], then slide by adding/removing one at a time.
# TRAP: sum(arr[i:i+k]) inside the loop re-slices → O(n·k).

Problem 2: Longest Substring Without Repeating Characters (Variable Window)

Given a string, find the length of the longest substring without repeating characters.

longest_unique_substring.py 
# LAYER 2: Variable Sliding Window — O(n) time, O(min(n, alphabet)) space
def length_of_longest_substring(s: str) -> int:
    seen = {}            # char → most recent index
    left = best = 0

    for right, ch in enumerate(s):
        if ch in seen and seen[ch] >= left:
            # Shrink window: jump left past the duplicate
            left = seen[ch] + 1
        seen[ch] = right
        best = max(best, right - left + 1)
    return best

print(length_of_longest_substring("abcabcbb"))   # 3  ("abc")
print(length_of_longest_substring("pwwkew"))     # 3  ("wke")

# LAYER 3: Pattern — Variable Sliding Window (expand/shrink)
# Template: right pointer always moves forward (expand)
#           left pointer jumps forward when condition breaks (shrink)
# Trigger: "longest/shortest substring/subarray with property X"
# Key: dict stores last seen index, enables O(1) jump for left ptr

Problem 3: Minimum Window Substring (Hard — Variable Window + Frequency)

Given strings s and t, find the minimum window in s that contains all characters of t.

minimum_window_substring.py 
from collections import Counter, defaultdict

# O(n) time, O(t) space — the hardest sliding window template
def min_window(s: str, t: str) -> str:
    need = Counter(t)
    required = len(need)          # unique chars needed
    window_counts = defaultdict(int)
    left = formed = 0
    result = (float('inf'), 0, 0)   # (length, left, right)

    for right, c in enumerate(s):
        window_counts[c] += 1

        # Check if this char's frequency requirement is met
        if c in need and window_counts[c] == need[c]:
            formed += 1

        # Contract window while all chars are satisfied
        while formed == required:
            if right - left + 1 < result[0]:
                result = (right - left + 1, left, right)
            left_char = s[left]
            window_counts[left_char] -= 1
            if left_char in need and window_counts[left_char] < need[left_char]:
                formed -= 1
            left += 1

    length, lo, hi = result
    return "" if length == float('inf') else s[lo:hi + 1]

print(min_window("ADOBECODEBANC", "ABC"))   # BANC
print(repr(min_window("a", "aa")))         # ''

# LAYER 3: Pattern — Variable Window + Frequency Counter
# This is the "master template" for sliding window problems.
# Expand right, track condition, contract left when condition met.
# Trade-off: 2 dicts (Counter + defaultdict) + 2 counters for O(n) time.
# NOTE: `c in need` isn't strictly required — a Counter returns 0 for a
#       missing key (without inserting it) — but it keeps the intent explicit.

Binary Search Pattern

Binary search is more than "find element in sorted array." The real power: find the boundary — the first/last position where a condition changes from False to True. This solves problems you wouldn't recognize as binary search.

PATTERN TRIGGER Sorted input, "find minimum/maximum that satisfies X", "search space can be halved" → Binary Search.

Problem 1: Binary Search — The Universal Template

Memorize this. Every binary search problem is a variation of this template. Python also ships the boundary version in the standard library: bisect.bisect_left / bisect_right.

binary_search_template.py 
import bisect

# THE TEMPLATE — O(log n) time, O(1) space
def binary_search(arr, target):
    lo, hi = 0, len(arr) - 1

    while lo <= hi:
        mid = lo + (hi - lo) // 2   # // = integer (floor) division
        if arr[mid] == target:
            return mid
        if arr[mid] < target:
            lo = mid + 1
        else:
            hi = mid - 1
    return -1

arr = [1, 3, 5, 7, 9]
print(binary_search(arr, 7))      # 3
print(binary_search(arr, 4))      # -1

# STANDARD LIBRARY: first index i with arr[i] >= x (insertion point)
print(bisect.bisect_left(arr, 5))   # 2
print(bisect.bisect_left(arr, 4))   # 2
print(bisect.bisect_right(arr, 5))  # 3 — first index with arr[i] > x

# KEY: use `//`, not `/` — `/` returns a float and arr[2.0] raises TypeError.
# Python ints never overflow, so (lo + hi) // 2 is safe here, but
# lo + (hi - lo) // 2 is the habit that transfers to Java/C++ interviews.

Problem 2: Search in Rotated Sorted Array

Array was sorted then rotated at some pivot. Search for a target in O(log n).

search_rotated.py 
# O(log n) — one of the most important binary search variants
def search(nums, target):
    lo, hi = 0, len(nums) - 1

    while lo <= hi:
        mid = lo + (hi - lo) // 2
        if nums[mid] == target:
            return mid

        # Determine which half is sorted
        if nums[lo] <= nums[mid]:
            # Left half is sorted
            if nums[lo] <= target < nums[mid]:   # chained comparison
                hi = mid - 1  # target in sorted left half
            else:
                lo = mid + 1  # target in right half
        else:
            # Right half is sorted
            if nums[mid] < target <= nums[hi]:
                lo = mid + 1  # target in sorted right half
            else:
                hi = mid - 1  # target in left half
    return -1

print(search([4, 5, 6, 7, 0, 1, 2], 0))   # 4
print(search([4, 5, 6, 7, 0, 1, 2], 3))   # -1

# KEY INSIGHT: in a rotated sorted array, at least one half
# is ALWAYS sorted. Check which half, then decide if target

Problem 3: Find Minimum in Rotated Sorted Array

Find the minimum element in a rotated sorted array (no duplicates).

find_min_rotated.py 
# O(log n) — "find the boundary" style binary search
def find_min(nums):
    lo, hi = 0, len(nums) - 1

    while lo < hi:
        mid = lo + (hi - lo) // 2
        if nums[mid] > nums[hi]:
            # Minimum is in the right half (rotation happened here)
            lo = mid + 1
        else:
            # Minimum is in the left half (including mid)
            hi = mid
    return nums[lo]

print(find_min([3, 4, 5, 1, 2]))      # 1
print(find_min([4, 5, 6, 7, 0, 1, 2])) # 0
