Union-Find Solutions

Components, cycles, merged groups and spanning trees — all with the same twenty-line DSU.

01Number of Connected Components in an Undirected GraphMEDIUM
Think: Begin with n separate components. Every edge that joins two different components reduces the count by one; an edge inside a component changes nothing. DSU's union tells you which case you're in.
1
Create a DSU with n nodes and components = n.
2
For each edge, if union(u, v) returns True, decrement components.
3
Return components.
Time: O(n + E · α(n))
Space: O(n)
SOLUTION
from typing import List


class DSU:
    def __init__(self, n: int):
        self.parent = list(range(n))
        self.size = [1] * n

    def find(self, x: int) -> int:
        while self.parent[x] != x:
            self.parent[x] = self.parent[self.parent[x]]   # path halving
            x = self.parent[x]
        return x

    def union(self, a: int, b: int) -> bool:
        ra, rb = self.find(a), self.find(b)
        if ra == rb:
            return False
        if self.size[ra] < self.size[rb]:
            ra, rb = rb, ra
        self.parent[rb] = ra
        self.size[ra] += self.size[rb]
        return True


def count_components(n: int, edges: List[List[int]]) -> int:
    dsu = DSU(n)
    components = n
    for u, v in edges:
        if dsu.union(u, v):       # merged two different groups
            components -= 1
    return components


print(count_components(5, [[0, 1], [1, 2], [3, 4]]))           # 2
print(count_components(5, [[0, 1], [1, 2], [2, 3], [3, 4]]))   # 1
print(count_components(3, []))                                 # 3
02Redundant ConnectionMEDIUM
Think: The graph is a tree plus one extra edge, and you must return the extra edge that appears last in the input. Add edges in order; the first edge whose endpoints are already connected closes the cycle — and because we scan in input order, it's the last cycle edge listed.
1
Nodes are labelled 1..n, so size the DSU n + 1 (index 0 unused).
2
For each edge, try union(u, v).
3
Return the first edge where union fails.
Time: O(n log n) worst case (path compression only)
Space: O(n)
SOLUTION — Compact Recursive DSU
from typing import List


def find_redundant_connection(edges: List[List[int]]) -> List[int]:
    parent = list(range(len(edges) + 1))   # nodes are 1..n, n == len(edges)

    def find(x: int) -> int:
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]

    for u, v in edges:
        ru, rv = find(u), find(v)
        if ru == rv:                        # already connected: this edge makes the cycle
            return [u, v]
        parent[ru] = rv
    return []


print(find_redundant_connection([[1, 2], [1, 3], [2, 3]]))                  # [2, 3]
print(find_redundant_connection([[1, 2], [2, 3], [3, 4], [1, 4], [1, 5]]))  # [1, 4]
03Graph Valid TreeMEDIUM
Think: A graph on n nodes is a tree iff it is connected and acyclic. Shortcut: a tree has exactly n - 1 edges. If the count is right and no edge closes a cycle, the n - 1 successful unions have merged n groups into 1, so it's connected too.
1
If len(edges) != n - 1, return False immediately.
2
Union every edge; if any union fails, there's a cycle → False.
3
Otherwise return True (n - 1 merges ⇒ one component).
Time: O(n · α(n))
Space: O(n)
SOLUTION

def valid_tree(n: int, edges: List[List[int]]) -> bool:
    if len(edges) != n - 1:          # too few → disconnected, too many → cycle
        return False
    dsu = DSU(n)
    return all(dsu.union(u, v) for u, v in edges)


print(valid_tree(5, [[0, 1], [0, 2], [0, 3], [1, 4]]))            # True
print(valid_tree(5, [[0, 1], [1, 2], [2, 3], [1, 3], [1, 4]]))    # False (cycle 1-2-3)
print(valid_tree(4, [[0, 1], [2, 3], [1, 0]]))                    # False (cycle, 2 parts)
04Accounts MergeMEDIUM
Think: Two accounts belong to the same person if they share any email — and that's transitive. So union account indices: remember the first account that owned each email, and when a later account lists the same email, union the two. Afterwards, group every email under its account's root.
1
owner dict: email → index of the first account containing it.
2
For account i, for each email: if already owned, union(i, owner[email]); else record it.
3
Bucket emails by find(owner[email]) using a defaultdict(list).
4
Each bucket becomes [name] + sorted(emails).
Time: O(N · α(n) + N log N), N = total emails
Space: O(N)
SOLUTION
from collections import defaultdict
from typing import Dict, List


class DSU:
    def __init__(self, n: int):
        self.parent = list(range(n))
        self.size = [1] * n

    def find(self, x: int) -> int:
        while self.parent[x] != x:
            self.parent[x] = self.parent[self.parent[x]]   # path halving
            x = self.parent[x]
        return x

    def union(self, a: int, b: int) -> bool:
        ra, rb = self.find(a), self.find(b)
        if ra == rb:
            return False
        if self.size[ra] < self.size[rb]:
            ra, rb = rb, ra
        self.parent[rb] = ra
        self.size[ra] += self.size[rb]
        return True


def accounts_merge(accounts: List[List[str]]) -> List[List[str]]:
    dsu = DSU(len(accounts))
    owner: Dict[str, int] = {}                    # email -> first account index
    for i, (_, *emails) in enumerate(accounts):   # skip the name, keep the emails
        for email in emails:
            if email in owner:
                dsu.union(i, owner[email])
            else:
                owner[email] = i

    groups: Dict[int, List[str]] = defaultdict(list)
    for email, i in owner.items():
        groups[dsu.find(i)].append(email)

    return [[accounts[root][0]] + sorted(emails) for root, emails in groups.items()]


accounts = [
    ["John", "johnsmith@mail.com", "john_newyork@mail.com"],
    ["John", "johnsmith@mail.com", "john00@mail.com"],
    ["Mary", "mary@mail.com"],
    ["John", "johnnybravo@mail.com"],
]
for merged in accounts_merge(accounts):
    print(merged)
# ['John', 'john00@mail.com', 'john_newyork@mail.com', 'johnsmith@mail.com']
# ['Mary', 'mary@mail.com']
# ['John', 'johnnybravo@mail.com']
05Number of ProvincesMEDIUM
Think: The input is an adjacency matrix: is_connected[i][j] == 1 means cities i and j are directly linked. A province is a connected component. The matrix is symmetric, so only look at the upper triangle (j > i) and union each linked pair, counting successful merges.
1
Start with provinces = n.
2
For each pair i < j with a 1 in the matrix, union them; decrement on a successful merge.
3
Return provinces. (DFS over the matrix also works in O(n²).)
Time: O(n² · α(n))
Space: O(n)
SOLUTION

def find_circle_num(is_connected: List[List[int]]) -> int:
    n = len(is_connected)
    dsu = DSU(n)
    provinces = n
    for i in range(n):
        for j in range(i + 1, n):             # symmetric matrix: upper triangle only
            if is_connected[i][j] and dsu.union(i, j):
                provinces -= 1
    return provinces


print(find_circle_num([[1, 1, 0], [1, 1, 0], [0, 0, 1]]))  # 2
print(find_circle_num([[1, 0, 0], [0, 1, 0], [0, 0, 1]]))  # 3
06Min Cost to Connect All Points (Kruskal)MEDIUM
Think: Points are vertices of a complete graph; the edge weight between two points is their Manhattan distance |x1 - x2| + |y1 - y2|. The cheapest way to connect everything is the MST. Kruskal: build all n(n-1)/2 edges, sort by weight, and add each edge that joins two different components until you've added n - 1 edges.
1
Generate every pair (i, j) with weight abs(xi - xj) + abs(yi - yj); itertools.combinations makes this one line.
2
Sort the edges by weight.
3
Union along each edge; on success add its weight and count it.
4
Stop at n - 1 edges and return the total.
Time: O(n² log n)
Space: O(n²) for the edge list
SOLUTION — Kruskal
from itertools import combinations
from typing import List


class DSU:
    def __init__(self, n: int):
        self.parent = list(range(n))
        self.size = [1] * n

    def find(self, x: int) -> int:
        while self.parent[x] != x:
            self.parent[x] = self.parent[self.parent[x]]   # path halving
            x = self.parent[x]
        return x

    def union(self, a: int, b: int) -> bool:
        ra, rb = self.find(a), self.find(b)
        if ra == rb:
            return False
        if self.size[ra] < self.size[rb]:
            ra, rb = rb, ra
        self.parent[rb] = ra
        self.size[ra] += self.size[rb]
        return True


def min_cost_connect_points(points: List[List[int]]) -> int:
    n = len(points)
    edges = sorted(
        (abs(x1 - x2) + abs(y1 - y2), i, j)
        for (i, (x1, y1)), (j, (x2, y2)) in combinations(enumerate(points), 2)
    )
    dsu = DSU(n)
    total = used = 0
    for w, i, j in edges:
        if dsu.union(i, j):
            total += w
            used += 1
            if used == n - 1:             # MST complete
                break
    return total


print(min_cost_connect_points([[0, 0], [2, 2], [3, 10], [5, 2], [7, 0]]))  # 20
print(min_cost_connect_points([[3, 12], [-2, 5], [-4, 1]]))               # 18
print(min_cost_connect_points([[0, 0]]))                                  # 0
ALTERNATIVE — Prim with heapq (no up-front edge list or sort)
import heapq
from typing import List


def min_cost_prim(points: List[List[int]]) -> int:
    n = len(points)
    in_tree = [False] * n
    heap = [(0, 0)]                       # (cost to attach, point index)
    total = added = 0
    while added < n:
        cost, i = heapq.heappop(heap)
        if in_tree[i]:
            continue                      # stale entry
        in_tree[i] = True
        total += cost
        added += 1
        xi, yi = points[i]
        for j in range(n):
            if not in_tree[j]:
                xj, yj = points[j]
                heapq.heappush(heap, (abs(xi - xj) + abs(yi - yj), j))
    return total


print(min_cost_prim([[0, 0], [2, 2], [3, 10], [5, 2], [7, 0]]))  # 20
